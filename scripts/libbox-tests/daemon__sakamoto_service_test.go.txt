package daemon_test

import (
    "context"
    "io"
    "net"
    "testing"
    "time"
    "github.com/sagernet/sing-box/adapter"
    "github.com/sagernet/sing-box/daemon"
    "github.com/sagernet/sing-box/include"
    M "github.com/sagernet/sing/common/metadata"
)

func TestSakamotoLiveLearningKeepsCoreAndTCPStream(t *testing.T) {
    ctx, cancel := context.WithCancel(include.Context(context.Background()))
    defer cancel()
    service := daemon.NewStartedService(daemon.ServiceOptions{Context: ctx, LogMaxLines: 20})
    t.Cleanup(func() { service.CloseService(); service.Close() })
    config := `{"log":{"disabled":true},"outbounds":[{"type":"direct","tag":"direct"}],"route":{"rule_set":[{"type":"inline","tag":"sakamoto-auto-proxy","rules":[{"domain_regex":["a^"]}]}],"rules":[{"rule_set":["sakamoto-auto-proxy"],"action":"route","outbound":"direct"}],"final":"direct"}}`
    if err := service.StartOrReloadService(ctx, config, nil); err != nil { t.Fatal(err) }
    if !service.IsServiceRunning() { t.Fatal("started core reported unavailable") }
    before := service.Instance()
    instanceID := service.RunningInstanceID()
    update := func(domains string) error { return service.UpdateAutomaticDomains(domains, instanceID) }
    listener, err := net.Listen("tcp", "127.0.0.1:0"); if err != nil { t.Fatal(err) }
    defer listener.Close()
    go func() { conn, err := listener.Accept(); if err == nil { defer conn.Close(); io.Copy(conn, conn) } }()
    client, inbound := net.Pipe()
    defer client.Close()
    defer inbound.Close()
    meta := adapter.InboundContext{Network:"tcp", Destination:M.ParseSocksaddr(listener.Addr().String()), Domain:"learned.example"}
    before.Box().Router().RouteConnectionEx(ctx, inbound, meta, nil)
    echo := func() {
        client.SetDeadline(time.Now().Add(3*time.Second))
        if _, err := client.Write([]byte("ssh-session")); err != nil { t.Fatal(err) }
        bytes := make([]byte, 11)
        if _, err := io.ReadFull(client, bytes); err != nil { t.Fatal(err) }
        if string(bytes) != "ssh-session" { t.Fatal("stream corrupted") }
    }
    echo()
    set, _ := before.Box().Router().RuleSet("sakamoto-auto-proxy")
    for range 5 {
        if err := update(`["learned.example"]`); err != nil { t.Fatal(err) }
        if !set.Match(&adapter.InboundContext{Domain:"learned.example"}) { t.Fatal("new domain was not applied") }
        if service.Instance() != before { t.Fatal("learning replaced the core") }
        echo()
        if err := update(`[]`); err != nil { t.Fatal(err) }
        if set.Match(&adapter.InboundContext{Domain:"learned.example"}) { t.Fatal("removed domain still matches") }
        echo()
    }
    if err := update(`["https://bad.example"]`); err == nil { t.Fatal("invalid domain accepted") }
    echo()
    if err := service.CloseService(); err != nil { t.Fatal(err) }
    if service.IsServiceRunning() { t.Fatal("stopped core reported running") }
    if err := update(`[]`); err == nil { t.Fatal("stopped core accepted learning") }
    client.SetDeadline(time.Now().Add(time.Second))
    if _, err := client.Write([]byte("closed")); err == nil { t.Fatal("stopped core retained the stream") }
    if err := service.StartOrReloadService(ctx, config, nil); err != nil { t.Fatal(err) }
    if err := update(`["stale.example"]`); err == nil { t.Fatal("old instance changed replacement rules") }
    replacement, _ := service.Instance().Box().Router().RuleSet("sakamoto-auto-proxy")
    if replacement.Match(&adapter.InboundContext{Domain:"stale.example"}) { t.Fatal("stale rule leaked into replacement") }
}
