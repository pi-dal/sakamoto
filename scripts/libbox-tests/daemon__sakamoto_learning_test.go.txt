type mobileOriginProber struct { server *daemon.StartedService; id int64 }
func (p mobileOriginProber) ProbeOrigin(domain string, port int32, outbound string, timeoutMillis int32) (int32, error) {
    return p.server.ProbeAutomaticOrigin(domain, port, outbound, p.id, timeoutMillis)
}

func TestSakamotoVerifiedLearningThroughNativeExits(t *testing.T) {
    var report []map[string]any
    for _, tc := range []struct { name string; direct, proxy int32; precedingRule bool; state string }{
        {"proxy_recovers", 502, 200, false, "verified"},
        {"direct_recovers", 200, 200, false, "direct_recovered"},
        {"both_fail", 502, 502, false, "not_improved"},
        {"denied", 502, 403, false, "not_improved"},
        {"rate_limited", 502, 429, false, "not_improved"},
        {"server_error", 502, 503, false, "not_improved"},
        {"explicit_direct_wins", 502, 200, true, "rolled_back"},
    } {
        t.Run(tc.name, func(t *testing.T) {
            var direct, proxy atomic.Int32
            direct.Store(tc.direct); proxy.Store(tc.proxy)
            dp := mobileOriginSOCKS(t, &direct); pp := mobileOriginSOCKS(t, &proxy)
            service := daemon.NewStartedService(daemon.ServiceOptions{Context: include.Context(context.Background()), LogMaxLines: 10})
            t.Cleanup(func() { service.CloseService(); service.Close() })
            prior := ""
            if tc.precedingRule { prior = `{"domain":["origin.example"],"action":"route","outbound":"direct"},` }
            config := fmt.Sprintf(`{"log":{"disabled":true},"outbounds":[{"type":"socks","tag":"direct","server":"127.0.0.1","server_port":%d},{"type":"socks","tag":"proxy","server":"127.0.0.1","server_port":%d}],"route":{"final":"direct","rule_set":[{"type":"inline","tag":"sakamoto-auto-proxy","rules":[{"domain_regex":["a^"]}]}],"rules":[%s{"rule_set":["sakamoto-auto-proxy"],"action":"route","outbound":"proxy"}]}}`, dp, pp, prior)
            if err := service.StartOrReloadService(context.Background(), config, nil); err != nil { t.Fatal(err) }
            prober := mobileOriginProber{service, service.RunningInstanceID()}
            check, err := mobileexperiment.NewLearningCheck(`{"domain":"origin.example","port":"80","failures":1,"required":3,"reason":"observed unmatched direct TCP dial failure"}`)
            if err != nil { t.Fatal(err) }
            check.Compare(prober, "proxy")
            if check.Ready() {
                if err := service.UpdateAutomaticDomains(`["origin.example"]`, prober.id); err != nil { t.Fatal(err) }
                if !check.Verify(prober) {
                    if err := service.UpdateAutomaticDomains(`[]`, prober.id); err != nil { t.Fatal(err) }
                }
            }
            var result map[string]any
            if err := json.Unmarshal([]byte(check.ReportJSON()), &result); err != nil { t.Fatal(err) }
            if result["state"] != tc.state { t.Fatalf("want %s: %s", tc.state, check.ReportJSON()) }
            if service.RunningInstanceID() != prober.id { t.Fatal("learning restarted core") }
            if _, err := prober.ProbeOrigin("https://bad.example", 80, "proxy", 1000); err == nil { t.Fatal("invalid domain accepted") }
            if _, err := prober.ProbeOrigin("origin.example", 22, "proxy", 1000); err == nil { t.Fatal("unsupported port accepted") }
            // A non-TLS proxy origin must never pass HTTPS validation.
            if _, err := prober.ProbeOrigin("origin.example", 443, "proxy", 1000); err == nil { t.Fatal("invalid TLS accepted") }
            if err := service.CloseService(); err != nil { t.Fatal(err) }
            if _, err := prober.ProbeOrigin("origin.example", 80, "proxy", 1000); err == nil { t.Fatal("stopped core accepted probe") }
            result["same_core"] = true
            report = append(report, result)
        })
    }
    if dir := os.Getenv("SAKAMOTO_TEST_ARTIFACT_DIR"); dir != "" {
        if err := os.MkdirAll(dir, 0700); err != nil { t.Fatal(err) }
        raw, _ := json.MarshalIndent(report, "", "  ")
        if err := os.WriteFile(filepath.Join(dir, "mobile-native-learning.json"), raw, 0600); err != nil { t.Fatal(err) }
    }
}

func TestSakamotoModeChangeStopsNativeProbe(t *testing.T) {
    // Hold the HTTP response so a real request is in flight at the switch.
    listener, err := net.Listen("tcp", "127.0.0.1:0")
    if err != nil { t.Fatal(err) }
    defer listener.Close()
    service := daemon.NewStartedService(daemon.ServiceOptions{Context: include.Context(context.Background())})
    defer func(){ service.CloseService(); service.Close() }()
    config := fmt.Sprintf(`{"log":{"disabled":true},"outbounds":[{"type":"socks","tag":"direct","server":"127.0.0.1","server_port":%d}],"route":{"rules":[{"clash_mode":"Global","action":"reject"}],"final":"direct"}}`, listener.Addr().(*net.TCPAddr).Port)
    if err := service.StartOrReloadService(context.Background(), config, nil); err != nil { t.Fatal(err) }
    prober := mobileOriginProber{service, service.RunningInstanceID()}
    done := make(chan error, 1)
    go func(){ _, err := prober.ProbeOrigin("origin.example", 80, "direct", 8000); done <- err }()
    listener.(*net.TCPListener).SetDeadline(time.Now().Add(2*time.Second))
    conn, err := listener.Accept(); if err != nil { t.Fatal(err) }; defer conn.Close()
    conn.SetDeadline(time.Now().Add(2*time.Second))
    var hello [2]byte
    if _, err := io.ReadFull(conn, hello[:]); err != nil { t.Fatal(err) }
    if _, err := io.CopyN(io.Discard,conn,int64(hello[1])); err != nil { t.Fatal(err) }
    if _, err := conn.Write([]byte{5,0}); err != nil { t.Fatal(err) }
    var head [5]byte
    if _, err := io.ReadFull(conn,head[:]); err != nil || head[3] != 3 { t.Fatal("expected domain dial",err) }
    if _, err := io.CopyN(io.Discard,conn,int64(head[4])+2); err != nil { t.Fatal(err) }
    if _, err := conn.Write([]byte{5,0,0,1,127,0,0,1,0,80}); err != nil { t.Fatal(err) }
    reader := bufio.NewReader(conn)
    request, err := http.ReadRequest(reader); if err != nil { t.Fatal(err) }; request.Body.Close()
    if _, err := service.SetClashMode(context.Background(), &daemon.ClashMode{Mode:"Global"}); err != nil { t.Fatal(err) }
    select { case err := <-done: if err == nil { t.Fatal("probe survived mode change") }; case <-time.After(time.Second): t.Fatal("probe was not canceled") }
    if _, err := reader.ReadByte(); err != io.EOF { t.Fatalf("in-flight connection did not close: %v", err) }
    if _, err := prober.ProbeOrigin("origin.example", 80, "direct", 500); err == nil { t.Fatal("new direct probe accepted") }
    listener.(*net.TCPListener).SetDeadline(time.Now().Add(100*time.Millisecond))
    if another, err := listener.Accept(); err == nil { another.Close(); t.Fatal("Global opened a direct socket") }
}

func TestSakamotoRegionalProbeIsolation(t *testing.T) {
    var code, directConnections, proxyConnections atomic.Int32
    code.Store(200)
    dp := mobileOriginSOCKS(t, &code, &directConnections)
    pp := mobileOriginSOCKS(t, &code, &proxyConnections)
    base := []byte(fmt.Sprintf(`{"log":{"disabled":true},"dns":{"servers":[{"type":"udp","tag":"ordinary","server":"127.0.0.1"}]},"route":{"final":"direct","rules":[]},"outbounds":[{"type":"socks","tag":"direct","server":"127.0.0.1","server_port":%d},{"type":"socks","tag":"proxy","server":"127.0.0.1","server_port":%d},{"type":"trojan","tag":"relay","server":"192.0.2.1","server_port":443,"password":"fixture","tls":{"enabled":true,"server_name":"relay.example"}}]}`, dp, pp))
    profile := regionaldns.Profile{Domains:"voice.example", Relay:"relay", ResolverIP:"192.0.2.2", TLSName:"fixture.example"}
    var err error
    profile.RelayFingerprint, err = regionaldns.RelayFingerprint(base, profile.Relay)
    if err != nil { t.Fatal(err) }
    config, err := regionaldns.Apply(base, profile, nil)
    if err != nil { t.Fatal(err) }
    service := daemon.NewStartedService(daemon.ServiceOptions{Context:include.Context(context.Background())})
    t.Cleanup(func(){ service.CloseService(); service.Close() })
    if err := service.StartOrReloadService(context.Background(), string(config), nil); err != nil { t.Fatal(err) }
    prober := mobileOriginProber{service, service.RunningInstanceID()}
    for _, exit := range []string{"direct", "proxy"} {
        if _, err := prober.ProbeOrigin("voice.example", 80, exit, 1000); err == nil { t.Error("protected origin probe succeeded") }
    }
    if directConnections.Load() != 0 || proxyConnections.Load() != 0 { t.Fatal("protected origin reached a learning exit") }
    for _, exit := range []string{"direct", "proxy"} {
        if status, err := prober.ProbeOrigin("control.example", 80, exit, 1000); err != nil || status != 200 { t.Fatalf("control probe failed: %d %v", status, err) }
    }
    if directConnections.Load() != 1 || proxyConnections.Load() != 1 { t.Fatal("control connection missing") }
    for _, candidate := range []struct { outbound map[string]any; reason string }{
        {map[string]any{"type":"wireguard", "tag":"wg", "peers":[]any{map[string]any{"address":"voice.example", "port":51820}}}, "proxy endpoint"},
        {map[string]any{"type":"hysteria2", "tag":"realm", "realm":map[string]any{"server_url":"https://control.example", "stun_servers":[]string{"VOICE.example.:3478"}}}, "proxy endpoint"},
        {map[string]any{"type":"socks", "tag":"extra", "server":"VOICE.example.", "server_port":443, "domain_resolver":"ordinary"}, "proxy endpoint"},
        {map[string]any{"type":"urltest", "tag":"extra", "outbounds":[]string{"proxy"}, "url":"https://VOICE.example.:443/probe"}, "latency test"},
    } {
        var imported map[string]any
        if err := json.Unmarshal(config, &imported); err != nil { t.Fatal(err) }
        if candidate.outbound["type"] == "wireguard" { imported["endpoints"] = []any{candidate.outbound} } else { imported["outbounds"] = append(imported["outbounds"].([]any), candidate.outbound) }
        raw, err := json.Marshal(imported)
        if err != nil { t.Fatal(err) }
        if err := service.StartOrReloadService(context.Background(), string(raw), nil); err == nil || !strings.Contains(err.Error(), candidate.reason) { t.Fatalf("unexpected admission result for %s: %v", candidate.reason, err) }
    }
    if directConnections.Load() != 1 || proxyConnections.Load() != 1 { t.Fatal("rejected configuration dialed a probe exit") }
    var stale map[string]any
    if err := json.Unmarshal(config, &stale); err != nil { t.Fatal(err) }
    servers := stale["dns"].(map[string]any)["servers"].([]any)
    servers[0].(map[string]any)["server"] = "VOICE.example."
    servers[0].(map[string]any)["domain_resolver"] = "regional-protected-doh"
    invalid, err := json.Marshal(stale)
    if err != nil { t.Fatal(err) }
    if err := service.StartOrReloadService(context.Background(), string(invalid), nil); err == nil || !strings.Contains(err.Error(), "overlaps a protected regional DNS name") { t.Fatalf("unexpected mobile startup result: %v", err) }
    if dir := os.Getenv("SAKAMOTO_TEST_ARTIFACT_DIR"); dir != "" {
        if err := os.MkdirAll(dir, 0700); err != nil { t.Fatal(err) }
        receipt, err := json.Marshal(map[string]any{"protected_direct_connections": 0, "protected_proxy_connections": 0, "control_connections": directConnections.Load() + proxyConnections.Load(), "bootstrap_conflict_rejected": true, "secondary_outbound_conflict_rejected": true, "latency_target_conflict_rejected": true})
        if err != nil { t.Fatal(err) }
        if err := os.WriteFile(filepath.Join(dir, "regional-native.json"), receipt, 0600); err != nil { t.Fatal(err) }
    }
}

func mobileOriginSOCKS(t *testing.T, code *atomic.Int32, connections ...*atomic.Int32) int {
    t.Helper()
    listener, err := net.Listen("tcp", "127.0.0.1:0"); if err != nil { t.Fatal(err) }
    var workers sync.WaitGroup
    var accepting sync.WaitGroup
    accepting.Add(1)
    t.Cleanup(func() { listener.Close(); accepting.Wait(); workers.Wait() })
    go func() {
        defer accepting.Done()
        for {
            conn, err := listener.Accept(); if err != nil { return }
            if len(connections) != 0 { connections[0].Add(1) }
            workers.Add(1)
            go func() {
                defer workers.Done(); defer conn.Close()
                conn.SetDeadline(time.Now().Add(2*time.Second))
                var hello [2]byte
                if _, err := io.ReadFull(conn, hello[:]); err != nil { return }
                if _, err := io.CopyN(io.Discard, conn, int64(hello[1])); err != nil { return }
                conn.Write([]byte{5,0})
                var request [4]byte
                if _, err := io.ReadFull(conn, request[:]); err != nil { return }
                n := 0
                switch request[3] {
                case 1: n = 4
                case 4: n = 16
                case 3:
                    var size [1]byte
                    if _, err := io.ReadFull(conn, size[:]); err != nil { return }; n = int(size[0])
                default: return
                }
                if _, err := io.CopyN(io.Discard, conn, int64(n+2)); err != nil { return }
                conn.Write([]byte{5,0,0,1,127,0,0,1,0,0})
                head, err := http.ReadRequest(bufio.NewReader(conn)); if err != nil { return }
                defer head.Body.Close()
                if head.Method != "HEAD" || head.URL.Path != "/" || head.Header.Get("Authorization") != "" || head.Header.Get("Cookie") != "" { t.Error("probe leaked app request details") }
                fmt.Fprintf(conn, "HTTP/1.1 %d Test\r\nContent-Length: 0\r\nConnection: close\r\n\r\n", code.Load())
            }()
        }
    }()
    return listener.Addr().(*net.TCPAddr).Port
}
